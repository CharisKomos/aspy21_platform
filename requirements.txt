# aspy21 0.2.0b15 is a pre-release. An exact == pin installs it without --pre.
# The wheel is pure Python (py3-none-any) and is available from both PyPI and
# piwheels; see README for the piwheels-specific install command.
aspy21==0.2.0b15

fastapi>=0.115
uvicorn[standard]>=0.30
jinja2>=3.1

# aspy21's own dependencies, pinned loosely and listed explicitly because the
# demo imports them directly.
httpx>=0.27
pandas>=2.0
tenacity>=9.0

# The mock HTTP layer. Only mock_backend.py uses it.
respx>=0.22

# Stores the live-mode password in the OS credential store (Windows Credential
# Manager) instead of a file. Only setup_live.py writes; config.py reads.
# The app degrades to ASPY21_PASSWORD if this is missing.
keyring>=25.0

# zoneinfo is stdlib, but Windows ships no IANA tz database for it to read. Only
# needed if ASPY21_TIMEZONE names a zone (Europe/Athens); 'UTC' and 'local' work
# without it. Harmless elsewhere, so it is not conditional.
tzdata>=2024.1

# Windows Integrated authentication, for sites where IIS fronts IP.21 with it and
# Basic just returns 401. Pure Python, so it installs anywhere; only used when
# ASPY21_AUTH_SCHEME=ntlm.
httpx-ntlm>=1.4

# Kerberos/SPNEGO (ASPY21_AUTH_SCHEME=negotiate) is deliberately NOT installed by
# default: httpx-gssapi needs system Kerberos libraries (libkrb5-dev to build,
# libkrb5-3 to run), which is a lot to impose on everyone for a scheme most sites
# do not use. Install it yourself where you need it:
#     pip install httpx-gssapi
# The app refuses to start with a clear message if you select the scheme without
